SELECT DISTINCT
    customer_id,
    customer_name,
    customer_email,
    customer_phone,
    customer_address
FROM `cal-2904-252b4aaa5753.wgu_d607_2c42895eccf52c7d.transactions`;

SELECT
    customer_name,
    transaction_id,
    product_id,
    vendor,
    product_category,
    product_name,
    quantity,
    price
FROM `cal-2904-252b4aaa5753.wgu_d607_2c42895eccf52c7d.transactions`
WHERE customer_name = 'Alexandra Young';

SELECT
    customer_id,
    customer_name,
    SUM(transaction_total) AS total_purchase_amount
FROM (
    SELECT DISTINCT
        transaction_id,
        customer_id,
        customer_name,
        total_amount AS transaction_total
    FROM `cal-2904-252b4aaa5753.wgu_d607_2c42895eccf52c7d.transactions`
)
GROUP BY customer_id, customer_name
ORDER BY total_purchase_amount DESC;
